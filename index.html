        function generateJournalDays(bulan, tahun, startDay = 1, endDay = null) {

            const daysInMonth = new Date(tahun, bulan, 0).getDate();

                let dayIdx = d.getDay();

                if (dayIdx === 0 || dayIdx === 6) continue;

                let holidayName = appState.holidayCache && appState.holidayCache[tahun] ? appState.holidayCache[tahun][isoDate] : null;

                let isHoliday = (holidayName !== null && holidayName !== undefined);

                    tanggal: `${i.toString().padStart(2, '0')}/${bulan.toString().padStart(2, '0')}/${tahun}`,

        }

        function angkaKeTerbilang(n) {

            n = Math.floor(n);

            if (n < 12) return satuan[n];

            if (n < 20) return angkaKeTerbilang(n - 10) + " belas";

            if (n < 100) return (satuan[Math.floor(n / 10)] || angkaKeTerbilang(Math.floor(n / 10))) + " puluh " + (n % 10 !== 0 ? " " + angkaKeTerbilang(n % 10) : "");

            if (n < 200) return "seratus" + (n % 100 !== 0 ? " " + angkaKeTerbilang(n - 100) : "");

            if (n < 1000) return angkaKeTerbilang(Math.floor(n / 100)) + " ratus" + (n % 100 !== 0 ? " " + angkaKeTerbilang(n % 100) : "");

            if (n < 2000) return "seribu" + (n % 1000 !== 0 ? " " + angkaKeTerbilang(n - 1000) : "");

            if (n < 10000) return angkaKeTerbilang(Math.floor(n / 1000)) + " ribu" + (n % 1000 !== 0 ? " " + angkaKeTerbilang(n % 1000) : "");

            return n.toString();

        }

        function calculatePenetapanDate(bulan, tahun) {

            const today = new Date();

            const targetBulan = parseInt(bulan);

            const targetTahun = parseInt(tahun);

            if (targetBulan === (today.getMonth() + 1) && targetTahun === today.getFullYear()) {

                const yyyy = today.getFullYear();

                const mm = (today.getMonth() + 1).toString().padStart(2, '0');

                const dd = today.getDate().toString().padStart(2, '0');

                return `${yyyy}-${mm}-${dd}`;

            }

            let d = new Date(targetTahun, targetBulan - 1, 1);

            let dayOfWeek = d.getDay();

            if (dayOfWeek === 6) d.setDate(3);

            else if (dayOfWeek === 0) d.setDate(2);

            

            const yyyy = d.getFullYear();

            const mm = (d.getMonth() + 1).toString().padStart(2, '0');

            const dd = d.getDate().toString().padStart(2, '0');

            return `${yyyy}-${mm}-${dd}`;

        }

        

        function resizeAndCompressImage(dataUrl, callback) {

            let img = new Image();

            img.onload = function() {

                let canvas = document.createElement('canvas');

                let ctx = canvas.getContext('2d');

                let maxW = 800; let maxH = 800;

                let w = img.width; let h = img.height;

                if(w > h) { if(w > maxW) { h *= maxW/w; w = maxW; } }

                else { if(h > maxH) { w *= maxH/h; h = maxH; } }

                canvas.width = w; canvas.height = h;

                ctx.drawImage(img, 0, 0, w, h);

                callback(canvas.toDataURL('image/jpeg', 0.7));

            };

            img.src = dataUrl;

        }

        function exportWord() {

            const element = $('doc-render-target');

            if(!element) return;

            

            const btn = $('btn-export-word');

            const originalText = btn.innerHTML;

            btn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin mr-1.5"></i> Memproses...';

            btn.disabled = true;

            

            const g = getActiveGuru();

            const fileName = `${activeDocType.toUpperCase()} ${getNamaBulan(appState.generator.bulan)} ${g.nama.split(',')[0].replace(/\s/g, ' ')}.doc`;

            

            const clone = element.cloneNode(true);

            const placeholders = clone.querySelectorAll('.upload-placeholder, input[type="file"]');

            placeholders.forEach(el => el.remove());

            

            const sectionDivs = clone.querySelectorAll('.SectionPortrait, .SectionLandscape');

            sectionDivs.forEach(el => {

                if (el.classList.contains('SectionLandscape')) {

                    el.className = 'SectionLandscape';

                } else {

                    el.className = 'SectionPortrait';

                }

            });

            

            const htmlContent = clone.innerHTML;

            

            const header = `

            <html xmlns:o='urn:schemas-microsoft-com:office:office'

                xmlns:w='urn:schemas-microsoft-com:office:word'

                xmlns='http://www.w3.org/TR/REC-html40'>

                <head>

                    <meta charset='utf-8'>

                    <title>Export Dokumen SertiGen</title>

                    <style>

                        ${getPageStyleHelper()}

                        body { font-family: Times New Roman; font-size: 11pt; color: black; }

                        table { border-collapse: collapse; width: 100%; }

                        th, td { padding: 4px; vertical-align: middle; }

                        .doc-table th, .doc-table td { border: 1pt solid windowtext; text-align: center; }

                        .doc-table th { font-weight: bold; background-color: #f8fafc; }

                        .info-table td { border: none; padding: 2pt; text-align: left; }

                        .doc-header-table td { border: none; }

                        .no-print-border { border: none !important; }

                        .doc-signatures td { border: none; }

                        .doc-sign-box { position: relative; text-align: center; }

                        .doc-sign-img { position: absolute; z-index: 2; height: 50px; width: auto; }

                        .doc-cap-img { position: absolute; z-index: 1; height: 65px; width: auto; opacity: 0.85; left: 10px; }

                        .word-section-break { display: block; clear: both; page-break-before: always; mso-break-type: section-break; }

                    </style>

                </head>

                <body>

                    ${htmlContent}

                </body>

            </html>`;

            

            const blob = new Blob(['\ufeff', header], { type: 'application/msword' });

            const url = URL.createObjectURL(blob);

            const link = document.createElement('a');

            link.href = url;

            link.download = fileName;

            document.body.appendChild(link);

            link.click();

            document.body.removeChild(link);

            URL.revokeObjectURL(url);

            

            btn.innerHTML = originalText;

            btn.disabled = false;

            showToast("Dokumen berhasil diexport ke Word.");

        }
